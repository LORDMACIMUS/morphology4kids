    /* App State */
    let currentSetKey = "set1";
    let currentUnitIndex = 0;
    let foundWords = new Set();

    /* Initialize Application */
    window.addEventListener('DOMContentLoaded', () => {
      renderMorphemeSelector();
      loadUnit(0);
    });

    /* Switch Grade Sets */
    function switchSet(setNum) {
      currentSetKey = `set${setNum}`;
      currentUnitIndex = 0;
      
      document.querySelectorAll('.tab-btn').forEach((btn, idx) => {
        btn.classList.toggle('active', idx === (setNum - 1));
      });

      document.getElementById('quizWorkspace').style.display = 'none';
      document.getElementById('lessonWorkspace').style.display = 'flex';

      renderMorphemeSelector();
      loadUnit(0);
    }

    /* Render Morpheme Pills */
    function renderMorphemeSelector() {
      const container = document.getElementById('morphemeSelector');
      container.innerHTML = '';
      
      CURRICULUM[currentSetKey].forEach((unit, idx) => {
        const pill = document.createElement('div');
        pill.className = `morpheme-pill ${idx === currentUnitIndex ? 'active' : ''}`;
        pill.textContent = unit.id;
        pill.onclick = () => loadUnit(idx);
        container.appendChild(pill);
      });
    }

    /* Load Selected Morpheme Unit */
    function loadUnit(index) {
      currentUnitIndex = index;
      const unit = CURRICULUM[currentSetKey][index];

      // Update Active Pill
      document.querySelectorAll('.morpheme-pill').forEach((pill, idx) => {
        pill.classList.toggle('active', idx === index);
      });

      // Update Unit Header & Overview
      document.getElementById('unitTitle').textContent = unit.id;
      document.getElementById('unitType').textContent = unit.type;
      document.getElementById('unitMeaning').textContent = `Meaning: ${unit.meaning}`;
      document.getElementById('unitInstruction').innerHTML = `
        The ${unit.type.toLowerCase()} <strong>${unit.id}</strong> means "<em>${unit.meaning}</em>". 
        Find words that build on this meaning!
      `;

      // Clear & Populate Tag Containers
      const exContainer = document.getElementById('examplesContainer');
      exContainer.innerHTML = '';
      unit.examples.slice(0, 4).forEach(w => addTag(w, 'example'));

      const nonExContainer = document.getElementById('nonExamplesContainer');
      nonExContainer.innerHTML = '';
      unit.nonExamples.forEach(w => addTag(w, 'nonexample'));

      // Populate Speed Fluency Grid (7-word repeating row)
      const fluencyContainer = document.getElementById('fluencyGrid');
      fluencyContainer.innerHTML = '';
      const gridWords = [...unit.examples, ...unit.examples].slice(0, 14);
      gridWords.forEach(w => {
        const cell = document.createElement('div');
        cell.className = 'fluency-cell';
        cell.textContent = w;
        cell.onclick = () => {
          speakOwly(w);
          cell.style.background = '#81B29A';
          cell.style.color = 'white';
        };
        fluencyContainer.appendChild(cell);
      });

      // Owly Introduction
      const introSpeech = `Hoot! Exploring ${unit.type} ${unit.id}, which means ${unit.meaning}.`;
      speakOwly(introSpeech);
    }

    /* Helper: Add Tag Pill */
    function addTag(word, type) {
      const container = type === 'example' ? document.getElementById('examplesContainer') : document.getElementById('nonExamplesContainer');
      const tag = document.createElement('span');
      tag.className = `tag tag-${type}`;
      tag.textContent = (type === 'example' ? '✓ ' : '⚠️ ') + word;
      container.appendChild(tag);
    }

    /* Submit & Evaluate Word Input */
    function submitWord() {
      const input = document.getElementById('wordInput');
      const word = input.value.trim().toLowerCase();
      if (!word) return;

      const unit = CURRICULUM[currentSetKey][currentUnitIndex];

      // Check if it's a known false root / non-example
      if (unit.nonExamples.includes(word)) {
        addTag(word, 'nonexample');
        const alertMsg = `Watch out! ${word} looks like it has ${unit.id}, but it is a false root!`;
        speakOwly(alertMsg);
      } 
      // Valid example match
      else if (unit.examples.includes(word) || word.includes(unit.id.replace(/[-]/g, ''))) {
        if (!foundWords.has(word)) {
          foundWords.add(word);
          addTag(word, 'example');
          updateProgress();
        }
        speakOwly(`Awesome! ${word} is a great example of ${unit.id}!`);
      } else {
        speakOwly(`Hmm, let's look closer at ${word}. Does it fit the meaning: ${unit.meaning}?`);
      }

      input.value = '';
    }

    /* Update Progress & Tier Badges */
    function updateProgress() {
      const count = foundWords.size;
      document.getElementById('wordsFoundCount').textContent = `${count} Words Discovered`;
      
      const bar = document.getElementById('unitProgressBar');
      const percentage = Math.min((count / 15) * 100, 100);
      bar.style.width = `${percentage}%`;

      const avatar = document.getElementById('owlyAvatar');
      const badge = document.getElementById('owlyTier');

      if (count >= 15) {
        badge.className = 'tier-badge tier-awesome';
        badge.textContent = 'AWESOME DETECTIVE! 👑';
        avatar.textContent = '🦉👑';
      } else if (count >= 10) {
        badge.className = 'tier-badge tier-amazing';
        badge.textContent = 'AMAZING DETECTIVE! 🔍';
        avatar.textContent = '🦉🔍';
      } else if (count >= 5) {
        badge.className = 'tier-badge tier-good';
        badge.textContent = 'GOOD JOB! ⭐';
        avatar.textContent = '🦉⭐';
      }
    }

    /* Web Speech Voice Engine (Owly Michael Ironside Voice) */
    function speakOwly(text) {
      document.getElementById('owlySpeech').textContent = text;
      
      if (!('speechSynthesis' in window)) return;
      window.speechSynthesis.cancel();

      const utter = new SpeechSynthesisUtterance(text);
      utter.pitch = parseFloat(document.getElementById('pitchSlider').value) || 0.3;
      utter.rate = parseFloat(document.getElementById('rateSlider').value) || 0.82;

      // Select deep/male voice if available
      const voices = window.speechSynthesis.getVoices();
      const maleVoice = voices.find(v => v.name.toLowerCase().includes('male') || v.name.toLowerCase().includes('david') || v.name.toLowerCase().includes('george'));
      if (maleVoice) utter.voice = maleVoice;

      const dot = document.getElementById('speakingDot');
      const avatar = document.getElementById('owlyAvatar');

      utter.onstart = () => {
        dot.classList.add('active');
        avatar.classList.add('bounce');
      };
      utter.onend = () => {
        dot.classList.remove('active');
        avatar.classList.remove('bounce');
      };

      window.speechSynthesis.speak(utter);
    }

      speakOwly("Hoot! I am Owly, your gravelly voice morphology guide!");
    }

    /* End of Set Mastery Quiz */
    function startSetQuiz() {
      document.getElementById('lessonWorkspace').style.display = 'none';
      const quizView = document.getElementById('quizWorkspace');
      quizView.style.display = 'block';

      const unit = CURRICULUM[currentSetKey][0];
      document.getElementById('quizTitle').textContent = `Mastery Quiz: ${currentSetKey.toUpperCase()}`;
      document.getElementById('quizQuestion').textContent = `Which of the following words correctly uses the morpheme "${unit.id}" (${unit.meaning})?`;

      const optionsContainer = document.getElementById('quizOptions');
      optionsContainer.innerHTML = '';

      const correctWord = unit.examples[0];
      const wrong1 = unit.nonExamples[0] || "uncle";
      const wrong2 = "table";

      const opts = [correctWord, wrong1, wrong2].sort(() => Math.random() - 0.5);

      opts.forEach(opt => {
        const btn = document.createElement('button');
        btn.className = 'quiz-opt-btn';
        btn.textContent = opt;
        btn.onclick = () => {
          if (opt === correctWord) {
            speakOwly(`Correct! ${opt} is a true example! You mastered this set!`);
            btn.style.background = '#81B29A';
            btn.style.color = 'white';
          } else {
            speakOwly(`Oops! ${opt} is a non-example or false root. Try again!`);
            btn.style.background = '#E07A5F';
            btn.style.color = 'white';
          }
        };
        optionsContainer.appendChild(btn);
      });

      speakOwly(`Welcome to the End of Set Quiz! Answer the question on screen.`);
    }
